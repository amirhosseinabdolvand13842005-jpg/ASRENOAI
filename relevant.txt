function previewReceipt(inp){
 if(!inp.files||!inp.files[0])return;
 receiptFile=inp.files[0];

 r.onload=e=>{$("receiptPreview").src=e.target.result;$("receiptPreview").classList.remove("hidden")};

}
function compressReceipt(file){
 return new Promise(resolve=>{
   if(!file){resolve("");return}
   const img=new Image(), reader=new FileReader();
   reader.onload=e=>img.src=e.target.result;
   img.onload=()=>{
     const max=950, scale=Math.min(1,max/Math.max(img.width,img.height)), c=document.createElement("canvas");
     c.width=Math.round(img.width*scale); c.height=Math.round(img.height*scale);
     c.getContext("2d").drawImage(img,0,0,c.width,c.height); resolve(c.toDataURL("image/jpeg",.72))
   };
   reader.readAsDataURL(file)
 })
}
